// 2. INITIALIZATION & PASSCODE LOGIC
// ==========================================
document.getElementById("gf-name").textContent = CONFIG.girlfriendName;

const lockScreen = document.getElementById("lock-screen");
const cakeScreen = document.getElementById("cake-screen");
const albumScreen = document.getElementById("album-screen");
const bgMusic = document.getElementById("bg-music");
const musicBtn = document.getElementById("music-toggle");

let isMusicPlaying = false;

function toggleMusic(forcePlay = false) {
  if (forcePlay || !isMusicPlaying) {
    bgMusic.volume = 0.4;
    bgMusic.play().then(() => {
      isMusicPlaying = true;
      musicBtn.textContent = "🔊";
    }).catch(() => {});
  } else {
    bgMusic.pause();
    isMusicPlaying = false;
    musicBtn.textContent = "🔇";
  }
}

musicBtn.addEventListener("click", () => toggleMusic(false));

document.getElementById("unlock-btn").addEventListener("click", () => {
  const entered = document.getElementById("passcode-input").value.trim();
  const errorMsg = document.getElementById("error-msg");

  if (entered === CONFIG.passcode) {
    lockScreen.classList.remove("active");
    cakeScreen.classList.add("active");
    musicBtn.classList.remove("hidden");
    toggleMusic(true);
    initMicrophone(); // Cake screen aate hi mic start karein
  } else {
    errorMsg.textContent = "Oops! Wrong date, try again ❤️";
  }
});

// 3. MICROPHONE BLOW DETECTION & CONFETTI
// ==========================================
let candleBlown = false;
let audioContext, analyser, microphone;

async function initMicrophone() {
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    audioContext = new (window.AudioContext || window.webkitAudioContext)();
    analyser = audioContext.createAnalyser();
    microphone = audioContext.createMediaStreamSource(stream);
    microphone.connect(analyser);
    analyser.fftSize = 256;

    const dataArray = new Uint8Array(analyser.frequencyBinCount);
    const micLevelBar = document.getElementById("mic-level");

    function detectBlow() {
      if (candleBlown) return;
      analyser.getByteFrequencyData(dataArray);

      // Low frequencies (phoonk ki awaz) ka average nikalna
      let sum = 0;
      for (let i = 0; i < 20; i++) {
        sum += dataArray[i];
      }
      let average = sum / 20;

      // Visual bar update
      micLevelBar.style.width = Math.min(100, (average / 160) * 100) + "%";

      // Threshold cross hone par candle bujha dein
      if (average > 135) {
        blowOutCandle();
      } else {
        requestAnimationFrame(detectBlow);
      }
    }
    detectBlow();
  } catch (err) {
    // Agar mic permission deny ho toh fallback button already screen par hai
    console.log("Mic access skipped, manual button available.");
  }
}

function blowOutCandle() {
  if (candleBlown) return;
  candleBlown = true;

  document.getElementById("flame").classList.add("out");
  document.getElementById("mic-controls").classList.add("hidden");
  document.getElementById("instruction-text").classList.add("hidden");
  document.getElementById("after-blow-box").classList.remove("hidden");

  // Confetti Celebration
  confetti({
    particleCount: 120,
    spread: 80,
    origin: { y: 0.6 }
  });
  setTimeout(() => {
    confetti({ particleCount: 80, angle: 60, spread: 55, origin: { x: 0 } });
    confetti({ particleCount: 80, angle: 120, spread: 55, origin: { x: 1 } });
  }, 400);
}

document.getElementById("manual-blow-btn").addEventListener("click", blowOutCandle);

// 4. SWIPEABLE PHOTO & VIDEO ALBUM
// ==========================================
let currentIndex = 0;
const mediaCard = document.getElementById("media-card");
const dotsContainer = document.getElementById("dots-container");

function renderDots() {
  dotsContainer.innerHTML = "";
  CONFIG.memories.forEach((_, idx) => {
    const dot = document.createElement("div");
    dot.className = "dot" + (idx === currentIndex ? " active" : "");
    dotsContainer.appendChild(dot);
  });
}

function renderMemory(index) {
  const item = CONFIG.memories[index];
  mediaCard.innerHTML = "";

  if (item.type === "video") {
    const video = document.createElement("video");
    video.src = item.src;
    video.controls = true;
    video.playsInline = true;
    video.autoplay = true;
    video.muted = true; // Mobile autoplay policy ke liye initially muted, user unmute kar sakta hai
    mediaCard.appendChild(video);
  } else {
    const img = document.createElement("img");
    img.src = item.src;
    img.alt = item.caption;
    mediaCard.appendChild(img);
  }

  const cap = document.createElement("p");
  cap.className = "polaroid-caption";
  cap.textContent = item.caption;

  const dt = document.createElement("p");
  dt.className = "polaroid-date";
  dt.textContent = item.date;

  mediaCard.appendChild(cap);
  mediaCard.appendChild(dt);
  renderDots();
}

document.getElementById("to-album-btn").addEventListener("click", () => {
  cakeScreen.classList.remove("active");
  albumScreen.classList.add("active");
  renderMemory(currentIndex);
});

document.getElementById("prev-btn").addEventListener("click", () => {
  currentIndex = (currentIndex - 1 + CONFIG.memories.length) % CONFIG.memories.length;
  renderMemory(currentIndex);
});

document.getElementById("next-btn").addEventListener("click", () => {
  currentIndex = (currentIndex + 1) % CONFIG.memories.length;
  renderMemory(currentIndex);
});

// Touch Swipe Support for Mobile
let touchStartX = 0;
mediaCard.addEventListener("touchstart", (e) => {
  touchStartX = e.changedTouches[0].screenX;
});

mediaCard.addEventListener("touchend", (e) => {
  let touchEndX = e.changedTouches[0].screenX;
  if (touchStartX - touchEndX > 40) {
    currentIndex = (currentIndex + 1) % CONFIG.memories.length;
    renderMemory(currentIndex);
  } else if (touchEndX - touchStartX > 40) {
    currentIndex = (currentIndex - 1 + CONFIG.memories.length) % CONFIG.memories.length;
    renderMemory(currentIndex);
  }
});
